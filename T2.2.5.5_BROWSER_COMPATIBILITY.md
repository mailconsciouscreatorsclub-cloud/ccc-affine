```typescript
// VoiceProviderFactory - Intelligent fallback
export class VoiceProviderFactory {
  public async selectProvider(): Promise<SpeechRecognitionProvider> {
    // Try Web Speech API first
    if (this.isWebSpeechAvailable()) {
      return this.webSpeechProvider;
    }

    // Firefox fallback: Use cloud provider
    if (this.isFirefox()) {
      console.warn('Firefox: Web Speech API not enabled, using cloud provider');
      return await this.getCloudProvider('google'); // Or Azure
    }

    throw new Error('No speech recognition provider available');
  }

  private isFirefox(): boolean {
    return navigator.userAgent.includes('Firefox');
  }
}
```

```typescript
private async loadVoices(): Promise<VoiceInfo[]> {

    if (voices.length > 0) {
```

## Browser Feature Detection

### Robust Feature Detection Code

```typescript
/**
 * Detect voice control capabilities across browsers
 */
export function getVoiceCapabilities(): {
  speechRecognition: boolean;
  speechSynthesis: boolean;
  mediaDevices: boolean;
  audioContext: boolean;
  browserName: string;
  browserVersion: string;
} {
  if (typeof window === 'undefined') {
    return {
      speechRecognition: false,
      speechSynthesis: false,
      mediaDevices: false,
      audioContext: false,
      browserName: 'SSR',
      browserVersion: '0.0.0',
    };
  }

  // Detect browser
  const ua = navigator.userAgent;
  let browserName = 'Unknown';
  if (ua.includes('Chrome') && !ua.includes('Edge')) browserName = 'Chrome';
  else if (ua.includes('Edg')) browserName = 'Edge';
  else if (ua.includes('Firefox')) browserName = 'Firefox';
  else if (ua.includes('Safari') && !ua.includes('Chrome')) browserName = 'Safari';

  return {
    speechRecognition:
      'SpeechRecognition' in window ||
      'webkitSpeechRecognition' in window,
    speechSynthesis:
      'speechSynthesis' in window,
    mediaDevices:
      'mediaDevices' in navigator &&
      'getUserMedia' in navigator.mediaDevices,
    audioContext:
      'AudioContext' in window ||
      'webkitAudioContext' in window,
    browserName,
    browserVersion: navigator.appVersion,
  };
}
```

### User-Friendly Capability Messaging

```typescript
/**
 * Generate user-friendly capability message
 */
export function getCapabilityMessage(): string {
  const caps = getVoiceCapabilities();

  if (!caps.speechRecognition) {
    if (caps.browserName === 'Firefox') {
      return 'Voice recognition requires enabling an experimental feature in Firefox. See our setup guide.';
    }
    return 'Voice recognition is not supported in your browser. Please use Chrome, Edge, or Safari.';
  }

  if (!caps.mediaDevices) {
    return 'Microphone access is required for voice control. Please grant microphone permissions.';
  }

  return 'Voice control is fully supported! 🎤';
}
```

## Polyfills & Fallbacks

### Speech Recognition Fallback

```typescript
// Automatically select best provider based on browser
export class VoiceProviderFactory {
  public async selectProvider(
    config: VoiceControlConfig
  ): Promise<SpeechRecognitionProvider> {
    const caps = getVoiceCapabilities();

    // Try Web Speech API first (Chrome, Edge, Safari)
    if (caps.speechRecognition) {
      return this.webSpeechProvider;
    }

    // Firefox fallback: Use Google Cloud Speech
    if (caps.browserName === 'Firefox') {
      console.info('Using Google Cloud Speech for Firefox');
      return await this.loadProvider('google-cloud');
    }

    // Ultimate fallback: Azure Speech Services
    console.info('Using Azure Speech Services fallback');
    return await this.loadProvider('azure-speech');
  }
}
```

### Speech Synthesis Polyfill

```typescript
// All modern browsers support synthesis - no polyfill needed!
// But we handle edge cases:

private ensureSynthesisReady(): Promise<void> {
  return new Promise((resolve) => {
    const synth = window.speechSynthesis;
    const voices = synth.getVoices();

    if (voices.length > 0) {
      resolve();
    } else {
      // Safari/Chrome async voice loading
      synth.addEventListener('voiceschanged', () => resolve(), { once: true });

      // Timeout fallback
      setTimeout(() => resolve(), 1000);
    }
  });
}
```
